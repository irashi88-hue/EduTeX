\makeatletter

%=============================================================================
% INTERNAL STATE
%=============================================================================

\newif\ifedutex@initialized

\edutex@initializedfalse

%=============================================================================
% INITIALIZATION
%=============================================================================

\newcommand{\edutex@initialize}{%

    \ifedutex@initialized

        \edutex@warning{Framework already initialized}

    \else

        \global\edutex@initializedtrue

        \edutex@info{EduTeX initialized}

    \fi

}

%=============================================================================
% LOAD MANIFEST
%=============================================================================

\newcommand{\edutex@loadmanifest}[1]{%

    \edutex@info{Loading manifest '#1'}

    \edutex@loadfile{#1}

}

%=============================================================================
% LOAD BOOK
%=============================================================================

\newcommand{\edutex@loadbook}[1]{%

    \edutex@info{Loading book '#1'}

    \edutex@loadmanifest{#1/manifest.tex}

}

\makeatother
