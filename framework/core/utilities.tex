\makeatletter

%-----------------------------------------------------------------------------
% Info Message
%-----------------------------------------------------------------------------

\newcommand{\edutex@info}[1]{%
    \typeout{[EduTeX] INFO: #1}%
}

%-----------------------------------------------------------------------------
% Warning Message
%-----------------------------------------------------------------------------

\newcommand{\edutex@warning}[1]{%
    \PackageWarning{EduTeX}{#1}%
}

%-----------------------------------------------------------------------------
% Error Message
%-----------------------------------------------------------------------------

\newcommand{\edutex@error}[2]{%
    \PackageError{EduTeX}{#1}{#2}%
}

%-----------------------------------------------------------------------------
% File Exists
%-----------------------------------------------------------------------------

\newcommand{\edutex@iffileexists}[3]{%
    \IfFileExists{#1}
    {#2}
    {#3}
}

%-----------------------------------------------------------------------------
% Load File
%-----------------------------------------------------------------------------

\newcommand{\edutex@loadfile}[1]{%
    \edutex@iffileexists
        {#1}
        {
            \edutex@info{Loading '#1'}
            \input{#1}
        }
        {
            \edutex@error
                {Cannot find file '#1'}
                {Check your installation or manifest configuration.}
        }
}

\makeatother
